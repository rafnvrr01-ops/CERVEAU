  function drawColonist(c) {
    const s = w2s(c.x, c.y);
    if (s.x < -30 || s.x > W + 30 || s.y < -55 || s.y > H + 30) return;
    const moving = Math.abs(c.targetX - c.x) > 2 || Math.abs(c.targetY - c.y) > 2;
    if (moving) c.walkPhase += 0.22;
    const legOffset = Math.sin(c.walkPhase) * 2;
    const scale = c.isChild ? 0.7 : 1;
    ctx.fillStyle = 'rgba(0,0,0,0.4)';
    ctx.beginPath(); ctx.ellipse(s.x, s.y + 2, 9 * scale, 3 * scale, 0, 0, Math.PI * 2); ctx.fill();

    ctx.fillRect(s.x + 1 * scale, s.y - 11 * scale - legOffset, 3 * scale, 11 * scale + legOffset);

    const armor = world.researched.armor;
    ctx.fillStyle = armor ? '#7a7a8a' : (c.isChild ? '#5a8040' : '#a83020');
